cmake_minimum_required(VERSION 3.25)
project(WanderScope VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SMTG_USE_STATIC_CRT ON CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
set(SMTG_RUN_VST_VALIDATOR ON CACHE BOOL "")
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES ON CACHE BOOL "" FORCE)
if(NOT VST3_SDK_ROOT)
    include(FetchContent)
    FetchContent_Declare(vst3sdk GIT_REPOSITORY https://github.com/steinbergmedia/vst3sdk.git
        GIT_TAG 3cdf9ca5d1f5b1b21e0a86832aa4abe55607bd96)
    FetchContent_MakeAvailable(vst3sdk)
else()
    add_subdirectory("${VST3_SDK_ROOT}" sdk)
endif()
if(NOT VST3_SDK_ROOT)
    set(VST3_SDK_ROOT "${vst3sdk_SOURCE_DIR}")
endif()
set(SDK_ROOT "${VST3_SDK_ROOT}")
set(public_sdk_SOURCE_DIR "${SDK_ROOT}/public.sdk")
if(MINGW)
    # This SDK utility is unrelated to the bridge and assumes MSVC's aligned allocator.
    get_target_property(sdk_sources sdk SOURCES)
    list(FILTER sdk_sources EXCLUDE REGEX "utility/dataexchange\\.cpp$")
    set_property(TARGET sdk PROPERTY SOURCES "${sdk_sources}")
    foreach(tool validator moduleinfotool)
        if(TARGET ${tool})
            target_link_options(${tool} PRIVATE -municode)
        endif()
    endforeach()
endif()
smtg_add_vst3plugin(Wander_scope Bridge.cpp Meter.h)
smtg_target_add_library_main(Wander_scope)
target_sources(Wander_scope PRIVATE "${SDK_ROOT}/public.sdk/source/main/moduleinit.cpp")
target_link_libraries(Wander_scope PRIVATE sdk)
if(MINGW)
    set_target_properties(Wander_scope PROPERTIES
        LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/VST3/${CMAKE_BUILD_TYPE}/Wander_scope.vst3/Contents/x86_64-win"
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/VST3/${CMAKE_BUILD_TYPE}/Wander_scope.vst3/Contents/x86_64-win")
    target_link_options(Wander_scope PRIVATE -static)
endif()
if(WIN32)
    target_link_libraries(Wander_scope PRIVATE ws2_32)
endif()
if(APPLE)
    smtg_target_set_bundle(Wander_scope BUNDLE_IDENTIFIER "com.wanderwave.scope" COMPANY_NAME "Wanderwave")
endif()
enable_testing()
add_executable(meter_tests MeterTests.cpp)
add_test(NAME meter_core COMMAND meter_tests)
add_executable(bridge_tests BridgeTests.cpp "${SDK_ROOT}/public.sdk/source/main/moduleinit.cpp")
target_link_libraries(bridge_tests PRIVATE sdk sdk_hosting)
if(APPLE)
    target_link_libraries(bridge_tests PRIVATE "-framework CoreFoundation")
endif()
if(WIN32)
    target_link_libraries(bridge_tests PRIVATE ws2_32)
endif()
add_test(NAME bridge_processor COMMAND bridge_tests)
